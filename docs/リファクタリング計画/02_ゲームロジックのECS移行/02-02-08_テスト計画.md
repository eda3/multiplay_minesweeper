### ゲームプレイシナリオテスト

```rust
#[test]
fn test_complete_game_scenario() {
    let (mut world, mut schedule) = setup_integration_test();
    
    // ボード初期化
    schedule.run(&mut world);
    
    // ゲームプレイをシミュレート
    let mut revealed_count = 0;
    let mut flagged_count = 0;
    let safe_cells = world.query_filtered::<&Position, Without<Mine>>().iter(&world).count();
    
    // ゲームの終了条件に達するまでプレイ
    while revealed_count < safe_cells && 
          world.resource::<GameState>().status == GameStatus::Playing {
        
        // 次にクリックするセルを選択（ヒューリスティック）
        let next_position = select_next_move(&world);
        
        // 選択したアクションに基づいて操作
        if should_flag(&world, next_position) {
            // フラグを立てる
            simulate_right_click(&mut world, next_position);
            flagged_count += 1;
        } else {
            // セルを公開
            simulate_click(&mut world, next_position);
            
            // 公開前の数を記録
            let before_count = revealed_count;
            
            // システム実行
            schedule.run(&mut world);
            
            // 新しく公開されたセル数を計算
            revealed_count = world.query_filtered::<&Position, With<Revealed>>()
                                 .iter(&world)
                                 .count();
            
            println!("公開したセル: ({}, {}), 新たに公開されたセル数: {}", 
                     next_position.x, next_position.y, revealed_count - before_count);
        }
        
        // ゲーム状態を確認
        let game_state = world.resource::<GameState>();
        if game_state.status != GameStatus::Playing {
            println!("ゲーム終了: {:?}", game_state.status);
            break;
        }
    }
    
    // ゲーム結果の検証
    let final_state = world.resource::<GameState>().status;
    println!("最終ゲーム状態: {:?}, 公開セル: {}/{}, フラグ: {}", 
             final_state, revealed_count, safe_cells, flagged_count);
    
    // ゲームが適切に終了したことを確認
    assert!(final_state == GameStatus::Won || final_state == GameStatus::Lost);
}
```

## まとめ

マインスイーパーのボードロジックをECSアーキテクチャに移行するプロジェクトでは、包括的なテスト戦略を実施しています。単体テストから統合テスト、パフォーマンステスト、エッジケーステストまで様々なレベルのテストにより、機能の正確性と性能要件の達成を確認しています。

特に連鎖的セル公開アルゴリズムの最適化は、明確な性能向上を示し、大きなボードサイズでも安定した動作を実現しています。インデックスキャッシュによる追加最適化で、さらなるパフォーマンス向上を達成し、極端な条件下でもシステムの堅牢性を確保しました。今後はUIとの統合やWebAssembly環境での検証に焦点を当て、品質をさらに向上させる予定です。
