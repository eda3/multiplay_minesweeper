#### 連鎖的なセル公開アルゴリズムの最適化

```rust
// 旧: 再帰的な実装（スタックオーバーフローのリスクあり）
fn reveal_connected_cells(
    row: usize,
    col: usize,
    board: &mut Board
) -> Result<(), JsValue> {
    // インデックスを計算
    let index = row * board.width + col;
    
    // 既に訪問済みなら終了
    if visited[index] {
        return Ok(());
    }
    
    // このセルを訪問済みとしてマーク
    visited[index] = true;
    
    // 周囲のセルを再帰的に処理
    for adj_row in ... {
        for adj_col in ... {
            reveal_connected_cells(adj_row, adj_col, board)?;
        }
    }
    
    Ok(())
}

// 新: 非再帰的（イテレーティブ）な実装
fn reveal_connected_cells_iterative(
    start_row: usize,
    start_col: usize,
    board: &mut Board
) -> Result<(), JsValue> {
    // キューを使って処理するセルを管理（事前に容量確保）
    let estimated_capacity = (board.width * board.height) / 4;
    let mut queue = VecDeque::with_capacity(estimated_capacity);
    
    // 処理済みセルを記録するセット（こちらも事前に容量確保）
    let mut visited = HashSet::with_capacity(estimated_capacity);
    
    // 開始セルをキューに追加
    queue.push_back((start_row, start_col));
    
    // インデックス計算をキャッシュするためのルックアップテーブル
    let mut row_offset_cache = Vec::with_capacity(board.height);
    for row in 0..board.height {
        row_offset_cache.push(row * board.width);
    }
    
    // イテレーティブに処理
    while let Some((row, col)) = queue.pop_front() {
        let index = row_offset_cache[row] + col;
        
        // 既に処理済みならスキップ
        if !visited.insert(index) {
            continue;
        }
        
        // このセルを処理
        // ...
        
        // 隣接セルをキューに追加
        // ...
    }
    
    Ok(())
}
```
